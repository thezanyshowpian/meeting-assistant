faster-whisper==1.2.1
av>=13
numpy>=1.24

# Stage 2 — phonetic matching for the glossary and the phonetic verification gate.
# Double Metaphone separates genuine ASR errors from meaning-changing edits far
# better than string similarity: it accepts "sequel"->"SQL" (string sim 0.67) and
# rejects "deadline"->"headline" (string sim 0.87). Both of which a fuzzy string
# match gets backwards.
metaphone>=0.6

# Interface. The UI only calls run_pipeline(), so this is the most swappable
# dependency in the project.
streamlit>=1.40
